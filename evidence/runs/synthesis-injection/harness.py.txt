class CapturingTransport:
 def __init__(self):self.inner=JsonTransport(timeout=30,max_bytes=100000);self.calls=[]
 def _send(self,request):
  call={'endpoint':request.full_url,'payload':json.loads(request.data)};self.calls.append(call)
  status,response=self.inner._send(request)
  call['http_status']=status
  if isinstance(response,dict):
   call['usage']=response.get('usage')
  return status,response
